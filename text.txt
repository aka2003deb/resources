#pragma once
#include <systemc.h>

class counter : public sc_module {
public:
    // Internal Registers
    sc_uint<8> CTRL;                     // Timer Control Register
    sc_uint<8> VALUE;                    // Timer Value Register (Read-only)
    sc_uint<8> COMPARE;                  // Timer Compare Register
    sc_uint<8> STATUS;                   // Timer Interrupt Status Register

    // Ports
    sc_in<sc_time> inputClock;           // Clock period input
    sc_in<bool> reset;                   // Reset signal (active high)
    sc_in<sc_uint<8>> address;           // Address (8-bit)
    sc_in<sc_uint<8>> data;              // Data (8-bit)
    sc_in<bool> read_enable;             // Read Enable signal
    sc_in<bool> write_enable;            // Write Enable signal

    sc_out<bool> interrupt0;             // Compare match interrupt (Active High Pulse)
    sc_out<bool> interrupt1;             // Overflow interrupt (Active High Pulse)

    // Constructor
    SC_HAS_PROCESS(counter);
    counter(sc_module_name name);
    
    // Processes
    void timer_process();                // Main timer process (event-based)
    void reset_func();                   // Reset operation
    void register_access();              // Register access

private:
    // Events for interrupt scheduling
    sc_event compare_event;
    sc_event overflow_event;
    sc_event clear_interrupts_event;

    // Helper methods
    void schedule_events();
    void handle_compare_event();
    void handle_overflow_event();
    void clear_interrupts();
};#include "counter.h"

counter::counter(sc_module_name name) : sc_module(name) {
    // Initialize values
    CTRL = 0;
    VALUE = 0;
    STATUS = 0;
    COMPARE = 0;
    interrupt0.write(false);
    interrupt1.write(false);

    // Register processes
    SC_METHOD(timer_process);
    sensitive << inputClock << reset << compare_event << overflow_event 
             << clear_interrupts_event;
    dont_initialize();

    SC_METHOD(reset_func);
    sensitive << reset;

    SC_METHOD(register_access);
    sensitive << write_enable << read_enable;
}

void counter::reset_func() {
    if (reset.read()) {
        // Reset all values to default
        CTRL = 0;
        VALUE = 0;
        STATUS = 0;
        interrupt0.write(false);
        interrupt1.write(false);
        // Cancel any pending events
        compare_event.cancel();
        overflow_event.cancel();
        clear_interrupts_event.cancel();
    }
}

void counter::timer_process() {
    if (reset.read()) return;

    // Check if we need to schedule new events
    if (inputClock.event()) {
        schedule_events();
        return;
    }

    // Handle compare event
    if (compare_event.triggered()) {
        handle_compare_event();
        return;
    }

    // Handle overflow event
    if (overflow_event.triggered()) {
        handle_overflow_event();
        return;
    }

    // Handle interrupt clearing
    if (clear_interrupts_event.triggered()) {
        clear_interrupts();
        return;
    }
}

void counter::schedule_events() {
    if (!CTRL[0]) return; // Timer not enabled

    sc_time period = inputClock.read();
    
    // Calculate time until compare match
    if (CTRL[1] && COMPARE > VALUE) { // Compare interrupt enabled
        sc_time time_to_compare = period * (COMPARE - VALUE);
        compare_event.notify(time_to_compare);
    }

    // Calculate time until overflow
    if (CTRL[2]) { // Overflow interrupt enabled
        sc_time time_to_overflow = period * (0xFF - VALUE);
        overflow_event.notify(time_to_overflow);
    }
}

void counter::handle_compare_event() {
    if (!CTRL[1]) return; // Compare interrupt not enabled
    
    // Update status and trigger interrupt
    STATUS[0] = 1;
    interrupt0.write(true);
    
    // Schedule interrupt clearing after one period
    clear_interrupts_event.notify(inputClock.read());
    
    // Update value (for accurate overflow calculation)
    VALUE = COMPARE;
    
    // Reschedule events if needed
    schedule_events();
}

void counter::handle_overflow_event() {
    if (!CTRL[2]) return; // Overflow interrupt not enabled
    
    // Update status and trigger interrupt
    STATUS[1] = 1;
    interrupt1.write(true);
    
    // Schedule interrupt clearing after one period
    clear_interrupts_event.notify(inputClock.read());
    
    // Reset counter
    VALUE = 0;
    
    // Reschedule events
    schedule_events();
}

void counter::clear_interrupts() {
    interrupt0.write(false);
    interrupt1.write(false);
}

void counter::register_access() {
    if (write_enable.read()) {
        switch (address.read()) {
        case 0x0: // Timer Control Register
            CTRL = data.read();
            // Reschedule events if timer was enabled/disabled
            if (data.read()[0] != CTRL[0]) {
                compare_event.cancel();
                overflow_event.cancel();
                schedule_events();
            }
            break;
        case 0x8: // Timer Compare Register
            COMPARE = data.read();
            // Reschedule compare event if changed
            compare_event.cancel();
            schedule_events();
            break;
        case 0xC: // Timer Interrupt Status Register
            // Write-one-to-clear logic
            if (data.read()[0]) STATUS[0] = 0;
            if (data.read()[1]) STATUS[1] = 0;
            break;
        }
    }
    if (read_enable.read()) {
        switch (address.read()) {
        case 0x0: cout << "CTRL: " << CTRL << endl; break;
        case 0x4: cout << "VALUE: " << VALUE << endl; break;
        case 0x8: cout << "COMPARE: " << COMPARE << endl; break;
        case 0xC: cout << "STATUS: " << STATUS << endl; break;
        }
    }
}#pragma once
#include <systemc.h>

#define COMPARE_VALUE 32     //set the compare value here

class tb_counter : public sc_module {
public:
    // Ports
    sc_out<sc_time> tb_inputClock;       // Clock period output
    sc_out<bool> tb_reset;               // Reset signal
    sc_out<sc_uint<8>> tb_address;       // Address
    sc_out<sc_uint<8>> tb_data;          // Data
    sc_out<bool> tb_read_enable;         // Read enable
    sc_out<bool> tb_write_enable;        // Write enable

    sc_in<bool> tb_interrupt0;           // Compare match interrupt
    sc_in<bool> tb_interrupt1;           // Overflow interrupt

    // Constructor
    SC_HAS_PROCESS(tb_counter);
    tb_counter(sc_module_name name);

    // Test process
    void test();
    
    // Ordinary member functions
    void register_write(sc_uint<8> address_var, sc_uint<8> data_var);
    void register_read(sc_uint<8> address_var);
};#include "tb_counter.h"

tb_counter::tb_counter(sc_module_name name): sc_module(name) {
    SC_THREAD(test); // Main test process
}

void tb_counter::register_write(sc_uint<8> address_var, sc_uint<8> data_var) {
    tb_address.write(address_var);
    tb_data.write(data_var);
    tb_write_enable.write(true);
    wait(SC_ZERO_TIME);
    tb_write_enable.write(false);
}

void tb_counter::register_read(sc_uint<8> address_var) {
    tb_address.write(address_var);
    tb_read_enable.write(true);
    wait(SC_ZERO_TIME);
    tb_read_enable.write(false);
}

void tb_counter::test() {
    cout << "Starting Event-Based Timer Testbench..." << endl;

    // Set clock period (10ns)
    tb_inputClock.write(sc_time(10, SC_NS));

    // Step 1: Reset the system
    tb_reset.write(true);
    wait(SC_ZERO_TIME);
    tb_reset.write(false);

    // Step 2: Configure Timer Control Register
    cout << "Configuring Timer Control Register..." << endl;
    register_write(0x0, 0x7); // Enable Timer, Enable CMP, Enable OV

    // Step 3: Set Compare Register
    cout << "Setting Compare Register to " << COMPARE_VALUE << endl;
    register_write(0x8, COMPARE_VALUE);

    // Step 4: Observe interrupts
    cout << "Observing Timer Operation..." << endl;

    for (int i = 0; i < 30; i++) {  // Reduced iterations since we're event-based
        wait(100, SC_NS); // Wait longer between checks

        if (tb_interrupt0.read()) {
            cout << "Compare Match Interrupt at " << sc_time_stamp() << endl;
            register_write(0xC, 0x1); // Clear interrupt
        }

        if (tb_interrupt1.read()) {
            cout << "Overflow Interrupt at " << sc_time_stamp() << endl;
            register_write(0xC, 0x2); // Clear interrupt
        }
    }

    cout << "Test Completed. Ending Simulation..." << endl;
    sc_stop();
}#pragma once
#include "counter.h"
#include "tb_counter.h"
