
#pragma once
#include <systemc.h>

class counter : public sc_module {
public:
    // Internal Registers
     sc_uint<8> CTRL;                     // Timer Control Register
     sc_uint<8> VALUE;                    // Timer Value Register (Read-only)
     sc_uint<8> COMPARE;                  // Timer Compare Register
     sc_uint<8> STATUS;                   // Timer Interrupt Status Register

    // Ports
    sc_in<bool> clock;                   // Clock signal
    sc_in<bool> reset;                   // Reset signal (active high)
    sc_in<sc_uint<8>> address;           // Address (8-bit)
    sc_in<sc_uint<8>> data;              // Data (8-bit)
    sc_in<bool> read_enable;             // Read Enable signal
    sc_in<bool> write_enable;            // Write Enable signal

    sc_out<bool> interrupt0;             // Compare match interrupt (Active High)
    sc_out<bool> interrupt1;             // Overflow interrupt (Active High)

    // Constructor
    SC_HAS_PROCESS(counter);
    counter(sc_module_name name);
    
    // Processes
    void counter_logic();                  // Timer logic
    void reset_func();                   //For handling reset operation
    void register_access();              //For modifying the register values

};
#include "counter.h"

counter::counter(sc_module_name name) : sc_module(name) {
    //initilaize the values before the simulation
    CTRL = 0;
    VALUE = 0;
    STATUS = 0;
    COMPARE = 0;
 
    // Register processes
    SC_METHOD(counter_logic);
    sensitive << clock.pos();  // Trigger on positive edge of the clock

    SC_METHOD(register_access);
    sensitive << write_enable << read_enable;

    SC_METHOD(reset_func);
    sensitive << reset;
}
void counter::reset_func() {
    if (reset.read()) {
        // Reset all values to default
        CTRL = 0;
        VALUE = 0;
        STATUS = 0;
        //COMPARE register value is not changed on reset
        interrupt0.write(false);
        interrupt1.write(false);
    }
}
void counter::counter_logic() {
        if (CTRL[0]) { // Check Timer Enable (bit 0 of CTRL)
            VALUE++; // Increment timer counter
      
            // Check for Compare Match Interrupt
            if (CTRL[1] && VALUE == 32) { // CMP bit (bit 1 of CTRL)
                STATUS[0] = 1; // Set Compare Match Interrupt status bit
                interrupt0.write(true);
            }

            // Check for Overflow Interrupt
            if (CTRL[2] && VALUE == 0xFF) { // OV bit (bit 2 of CTRL)
                STATUS[1] = 1; // Set Overflow Interrupt status bit
                interrupt1.write(true);
            }  
        }
}

// This is the internal register access mechanism
void counter::register_access() {
    if (write_enable.read()) {
        switch (address.read()) {
        case 0x0: CTRL = data.read();  break;
        case 0x8: COMPARE = data.read(); break;
        case 0xC: 
            // Write-one-to-clear logic
            if (data.read()[0]) {
                STATUS[0] = 0;
                interrupt0.write(false);
            }
            if (data.read()[1]) {
                STATUS[1] = 0;
                interrupt1.write(false);
            }
            break;
        }
    }
    if (read_enable.read()) {
        switch (address.read()) {
        case 0x0:cout << "The current value of CTRl register is= " << CTRL << endl;
            break;
        case 0x4:cout << "The current value of VALUE register is= " << VALUE << endl;
            break;
        case 0x8:cout << "The current value of COMPARE register is= " << COMPARE << endl;
            break;
        case 0xC:cout << "The current value of STATUS register is= " << STATUS << endl;
            break;
        }
    }
}
#pragma once
#include <systemc.h>

#define COMPARE_VALUE 32     //set the compare value here

class tb_counter : public sc_module {
public:
    // Ports
    sc_out<bool> tb_clock;                  // Clock signal
    sc_out<bool> tb_reset;                  // Reset signal
    sc_out<sc_uint<8>> tb_address;          // Address
    sc_out<sc_uint<8>> tb_data;             // Data
    sc_out<bool> tb_read_enable;            // Read enable
    sc_out<bool> tb_write_enable;           // Write enable

    sc_in<bool> tb_interrupt0;              // Compare match interrupt
    sc_in<bool> tb_interrupt1;              // Overflow interrupt

    // Constructor
    SC_HAS_PROCESS(tb_counter);
    tb_counter(sc_module_name name);

    // Test process
    void test();
    void clock_cycle();

    //Ordinary member functions
    void register_write(sc_uint<8> address_var, sc_uint<8> data_var);
    void register_read(sc_uint<8> address_var);
};
#include "tb_counter.h"

tb_counter::tb_counter(sc_module_name name): sc_module(name) {
    SC_THREAD(test);        // Main test process
    SC_THREAD(clock_cycle); // Clock generation process
}

void tb_counter::clock_cycle() {
    while (true) {
        tb_clock.write(false);
        wait(5, SC_NS);  // Half clock period
        tb_clock.write(true);
        wait(5, SC_NS);  // Half clock period
    }
}

void tb_counter::register_write(sc_uint<8> address_var, sc_uint<8> data_var) {
    tb_address.write(address_var);
    tb_data.write(data_var);
    tb_write_enable.write(true);
    wait(SC_ZERO_TIME); // Wait for the write operation to complete
    tb_write_enable.write(false);
}

void tb_counter::register_read(sc_uint<8> address_var) {
    tb_address.write(address_var);
    tb_read_enable.write(true);
    wait(SC_ZERO_TIME); // Wait for the read operation to complete
    tb_read_enable.write(false);
}

void tb_counter::test() {
    cout << "Starting Timer Testbench..." << endl;

    // Step 1: Reset the system
    tb_reset.write(true);
    wait(SC_ZERO_TIME);
    tb_reset.write(false);

    // Step 2: Configure Timer Control Register (Enable Timer, Enable Interrupts)
    cout << "Configuring Timer Control Register..." << endl;
    register_write(0x0, 0x7); // Enable Timer, Enable CMP, Enable OV

    // Step 3: Set Compare Register
    cout << "Setting Compare Register to " << COMPARE_VALUE << endl;
    register_write(0x8, COMPARE_VALUE); // Set compare value

    // Step 4: Observe interrupts during timer operation
    cout << "Observing Timer Operation for Compare and Overflow Interrupts..." << endl;

    for (int i = 0; i < 300; i++) {  // Simulate 300 clock cycles
        wait(10, SC_NS); // Wait for one clock cycle

        if (tb_interrupt0.read()) {
            cout << "Compare Match Interrupt Triggered at Time: " << sc_time_stamp() << endl;
            // Clear Compare Match Interrupt
            register_write(0xC, 0x1); // Clear Compare Match Interrupt (STATUS[0])
        }

        if (tb_interrupt1.read()) {
            cout << "Overflow Interrupt Triggered at Time: " << sc_time_stamp() << endl;
            // Clear Overflow Interrupt
            register_write(0xC, 0x2); // Clear Overflow Interrupt (STATUS[1])
        }
    }

    // Step 5: Verify register values
    cout << "At the end of the simulation...Reading Timer Value Register..." << endl;
    register_read(0x4);
    cout << "Test Completed. Ending Simulation..." << endl;
    sc_stop(); // End simulation
}

Registers
The module should also consists of couple of 8bit Special Function Registers(SFRs) to configure the Timer module. The register sets as defined as below:
 
2.	Timer control Register(CTRL) - Offset Address = 0x0
 
 
0bit - En - Timer Enable bit, where if you configure it to:
0 => Timer is disable and counter does not increment All register can still be read and written 
1 => Timer is enabled and counter will increment normally. Timer counter will be incremented every clock cycle.
 
1bit - CMP- Timer compare Interrupt Enable, where if you configure it to:
0 =>Compare interrupt is disabled  
1 => Compare interrupt is enable. 
If enabled then whenever timer counter value reaches to compare value it will raise the interrupt line(Interrupt0).
 
2bit - OV - Overflow Interrupt Enable, where if you configure it to:
0 =>Overflow interrupt is disabled  
1 => Overflow interrupt is enable. 
if timer counter value reaches to OxFF, in the next clock cycle it will update start counting from zero, and raise the interrupt line(Interrupt1) corresponding to overflow interrupt.
 
3.	Timer Value Register - Offset Address = 0x4
 
 
This is a read only register, Whenever timer is enabled it will start incrementing value of this register. 
 
4.	Timer Compare Register - Offset Address = 0x8
 
 
This register can be used to store the comparison value for timer counter register. If timer compare interrupt is enable, then it will assert an interrupt line INTO whenever timer value matches with the timer compare value.
 
5.	Timer Interrupt status Register - Offset Address = 0xC
 
 
This is a write one clear interrupt status register. If timer is enabled then whenever timer counter value reaches a comparator value or an overflow occurs then it will update the corresponding status bit if related interrupt is enabled. Writing a 1 will clear the corresponding interrupt
